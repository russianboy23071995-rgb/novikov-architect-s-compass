export type Vec3 = [number, number, number];
export type Face = { wallId: string; vertices: Vec3[]; normal: Vec3 };
export type Solid = { faces: Face[]; volume: number; min: Vec3; max: Vec3 };

/** Partition the wall plane at opening edges, then extrude only occupied cells.
 * Emit boundary faces only: openings pass through the entire wall thickness.
 * Overlapping openings are subtracted as a union, not counted twice.
 */
export function buildSolid(input: Project): Solid {
  const project = validateProject(input);
  return buildPreviewSolid(project, workingStoreyElevation(project));
}

/** Derived geometry from a validated project or a validated prepared preview scope. */
export function buildPreviewSolid(project: ModelGeometry, elevation: number): Solid {
  return deriveSolid(project, elevation);
}
type Storey = Project["storeys"][number];
type WallResources = {
  prior: Storey;
  elevation: number;
  faces: Map<string, Face[]>;
  reused: Set<string>;
};
const extrusionKey = (
  wall: Storey["walls"][number],
  windows: Storey["windows"],
  elevation: number,
  shifted: boolean,
) => [
  wall.start.x,
  wall.start.y,
  wall.end.x,
  wall.end.y,
  wall.thickness,
  wall.height,
  wall.bodyOffset,
  wallBase(wall, elevation),
  Number(shifted),
  ...windows.flatMap((w) => [w.width, w.height, w.sillHeight, w.position]),
];
function deriveSolid(project: ModelGeometry, elevation: number, resources?: WallResources): Solid {
  const solid: Solid = {
    faces: [],
    volume: 0,
    min: [Infinity, Infinity, Infinity],
    max: [-Infinity, -Infinity, -Infinity],
  };
  const joined = new Map(connectedWallSolids(project).map((w) => [w.wallId, w]));
  // Derivation-local index: preserve window order and objects, never retain a
  // stale host assignment across model/preview changes.
  const windowsByHost = new Map<string, Project["storeys"][number]["windows"]>();
  for (const window of storeyOf(project).windows) {
    const group = windowsByHost.get(window.wallId);
    if (group) group.push(window);
    else windowsByHost.set(window.wallId, [window]);
  }
  const priorWalls = resources && new Map(resources.prior.walls.map((w) => [w.id, w]));
  const priorWindows = resources && new Map<string, Storey["windows"]>();
  const priorJoined =
    resources && new Set(connectedWallSolids({ storey: resources.prior }).map((w) => w.wallId));
  const shifted = !!resources && storeyOf(project).walls.some((w) => wallBase(w, elevation) !== 0);
  const priorShifted =
    !!resources && resources.prior.walls.some((w) => wallBase(w, resources.elevation) !== 0);
  if (resources)
    for (const opening of resources.prior.windows) {
      const group = priorWindows!.get(opening.wallId);
      if (group) group.push(opening);
      else priorWindows!.set(opening.wallId, [opening]);
    }
  for (const wall of storeyOf(project).walls) {
    const connection = joined.get(wall.id);
    if (connection) {
      solid.faces.push(...connection.faces);
      solid.volume += connection.volume;
      for (const face of connection.faces)
        for (const p of face.vertices)
          for (const axis of [0, 1, 2] as const) {
            solid.min[axis] = Math.min(solid.min[axis], p[axis]);
            solid.max[axis] = Math.max(solid.max[axis], p[axis]);
          }
      continue;
    }
    const length = wallLength(wall);
    const body = wallBody(wall);
    const ux = (wall.end.x - wall.start.x) / length;
    const uy = (wall.end.y - wall.start.y) / length;
    const openings = (windowsByHost.get(wall.id) ?? []).map((w) => ({
      left: w.position * length - w.width / 2,
      right: w.position * length + w.width / 2,
      bottom: w.sillHeight,
      top: w.sillHeight + w.height,
    }));
    const previous = priorWalls?.get(wall.id);
    let reused: Face[] | undefined;
    if (previous && !priorJoined!.has(wall.id)) {
      const before = extrusionKey(
        previous,
        priorWindows!.get(wall.id) ?? [],
        resources!.elevation,
        priorShifted,
      );
      const after = extrusionKey(wall, windowsByHost.get(wall.id) ?? [], elevation, shifted);
      if (before.length === after.length && before.every((v, i) => Object.is(v, after[i]))) {
        reused = resources!.faces.get(wall.id);
        if (reused) {
          resources!.reused.add(wall.id);
          for (const face of reused) {
            solid.faces.push(face);
            for (const point of face.vertices)
              for (const axis of [0, 1, 2] as const) {
                solid.min[axis] = Math.min(solid.min[axis], point[axis]);
                solid.max[axis] = Math.max(solid.max[axis], point[axis]);
              }
          }
        }
      }
    }
    const sorted = (values: number[]) => [...new Set(values)].sort((a, b) => a - b);
    const xs = sorted([0, length, ...openings.flatMap((w) => [w.left, w.right])]);
    const zs = sorted([0, wall.height, ...openings.flatMap((w) => [w.bottom, w.top])]);
    const occupied = xs.slice(1).map((right, i) =>
      zs.slice(1).map(
        (top, j) =>
          !openings.some((w) => {
            const x = (xs[i]! + right) / 2,
              z = (zs[j]! + top) / 2;
            return x > w.left && x < w.right && z > w.bottom && z < w.top;
          }),
      ),
    );
    const world = ([x, y, z]: Vec3): Vec3 => [
      body.start.x + ux * x - uy * y,
      body.start.y + uy * x + ux * y,
      z,
    ];
    const face = (points: [Vec3, Vec3, Vec3, Vec3], normal: Vec3) => {
      const vertices = points.map(world) as Face["vertices"];
      for (const p of vertices)
        for (const axis of [0, 1, 2] as const) {
          solid.min[axis] = Math.min(solid.min[axis], p[axis]);
          solid.max[axis] = Math.max(solid.max[axis], p[axis]);
        }
      solid.faces.push({
        wallId: wall.id,
        vertices,
        normal: [ux * normal[0] - uy * normal[1], uy * normal[0] + ux * normal[1], normal[2]],
      });
    };
    const a = -wall.thickness / 2,
      b = wall.thickness / 2;
    for (let i = 0; i < xs.length - 1; i++)
      for (let j = 0; j < zs.length - 1; j++) {
        if (!occupied[i]?.[j]) continue;
        const l = xs[i]!,
          r = xs[i + 1]!,
          d = zs[j]!,
          u = zs[j + 1]!;
        solid.volume += (r - l) * (u - d) * wall.thickness;
        if (reused) continue;
        face(
          [
            [l, a, d],
            [r, a, d],
            [r, a, u],
            [l, a, u],
          ],
          [0, -1, 0],
        );
        face(
          [
            [r, b, d],
            [l, b, d],
            [l, b, u],
            [r, b, u],
          ],
          [0, 1, 0],
        );
        if (!occupied[i - 1]?.[j])
          face(
            [
              [l, b, d],
              [l, a, d],
              [l, a, u],
              [l, b, u],
            ],
            [-1, 0, 0],
          );
        if (!occupied[i + 1]?.[j])
          face(
            [
              [r, a, d],
              [r, b, d],
              [r, b, u],
              [r, a, u],
            ],
            [1, 0, 0],
          );
        if (!occupied[i]?.[j - 1])
          face(
            [
              [l, b, d],
              [r, b, d],
              [r, a, d],
              [l, a, d],
            ],
            [0, 0, -1],
          );
        if (!occupied[i]?.[j + 1])
          face(
            [
              [l, a, u],
              [r, a, u],
              [r, b, u],
              [l, b, u],
            ],
            [0, 0, 1],
          );
      }
  }
  if (!solid.faces.length) {
    solid.min = [0, 0, 0];
    solid.max = [1, 1, 1];
  }
  const bases = new Map(storeyOf(project).walls.map((w) => [w.id, wallBase(w, elevation)]));
  if (solid.faces.length && [...bases.values()].some((z) => z !== 0)) {
    solid.faces = solid.faces.map((f) =>
      resources?.reused.has(f.wallId)
        ? f
        : {
            ...f,
            vertices: f.vertices.map(([x, y, z]): Vec3 => [x, y, z + bases.get(f.wallId)!]),
          },
    );
    solid.min[2] = Infinity;
    solid.max[2] = -Infinity;
    for (const face of solid.faces)
      for (const p of face.vertices) {
        solid.min[2] = Math.min(solid.min[2], p[2]);
        solid.max[2] = Math.max(solid.max[2], p[2]);
      }
  }
  return solid;
}

export type Camera = OrthographicCamera;
export const initialCamera: Camera = { yaw: -0.45, pitch: 0.3, zoom: 1, panX: 0, panY: 0 };

/** Orthographic camera in a Z-up world; depth is preserved for WebGL occlusion. */
export function projectPoint(point: Vec3, solid: Solid, camera: Camera, aspect: number): Vec3 {
  return projectOrthographic(point, createProjectionFrame(solid), camera, aspect);
}

type BuildingEntry = { project: Project; result: Solid };
const buildings = new WeakMap<Project, BuildingEntry>();
// Only one weak recent candidate: no project-ID registry or retained scene history.
let recentBuilding: WeakRef<BuildingEntry> | null = null;
function freezeBuilding(result: Solid): Solid {
  for (const face of result.faces) {
    for (const point of face.vertices) Object.freeze(point);
    Object.freeze(face.vertices);
    Object.freeze(face.normal);
    Object.freeze(face);
  }
  Object.freeze(result.faces);
  Object.freeze(result.min);
  Object.freeze(result.max);
  return Object.freeze(result);
}
function freezeResourceBuilding(result: Solid, reused: ReadonlySet<string>): Solid {
  // Freeze only freshly derived faces through the original deep-freeze routine.
  // Shared faces come exclusively from an already deeply frozen building result.
  freezeBuilding({ ...result, faces: result.faces.filter((face) => !reused.has(face.wallId)) });
  Object.freeze(result.faces);
  return Object.freeze(result);
}
/** One derived building scene. Storeys remain authoritative; working selection is not a filter. */
export function buildBuildingSolid(input: Project): Solid {
  const cached = buildings.get(input);
  const immutable = isOwnedProjectSnapshot(input);
  if (cached && immutable) return cached.result;
  // Mutable public inputs always pass full validation, including same-identity calls.
  const project = immutable ? input : validateProject(input);
  if (cached && sameProjectModel(project, cached.project)) return cached.result;
  const recent = recentBuilding?.deref();
  // Only factory-owned snapshots may reuse their prior validation.
  // Conservative model equality includes geometry, storey heights, datum and joins.
  if (recent && sameProjectModel(project, recent.project)) {
    buildings.set(input, recent);
    return recent.result;
  }
  // Only adjacent owned snapshots with identical height context enter the pilot.
  // Cold load and changed datum/storey heights use the unmodified full path.
  const canReuse =
    immutable &&
    recent &&
    recent.result.faces.length >= 1024 &&
    isOwnedProjectSnapshot(recent.project) &&
    project.id === recent.project.id &&
    Object.is(project.groundDatum, recent.project.groundDatum) &&
    project.groundStoreyId === recent.project.groundStoreyId &&
    project.storeys.length === recent.project.storeys.length &&
    project.storeys.every(
      (s, i) =>
        s.id === recent.project.storeys[i]!.id &&
        Object.is(s.height, recent.project.storeys[i]!.height),
    );
  const priorFaces = new Map<string, Face[]>();
  const reused = new Set<string>();
  if (canReuse)
    for (const face of recent.result.faces) {
      const group = priorFaces.get(face.wallId);
      if (group) group.push(face);
      else priorFaces.set(face.wallId, [face]);
    }
  const parts = project.storeys
    .filter((s) => s.walls.length > 0)
    .map((s) =>
      deriveSolid(
        { storey: s, assets: project.assets },
        workingStoreyElevation({ ...project, workingStoreyId: s.id }),
        canReuse
          ? {
              prior: recent.project.storeys.find((old) => old.id === s.id)!,
              elevation: workingStoreyElevation({ ...recent.project, workingStoreyId: s.id }),
              faces: priorFaces,
              reused,
            }
          : undefined,
      ),
    );
  const result: Solid = {
    faces: parts.flatMap((p) => p.faces),
    volume: parts.reduce((sum, p) => sum + p.volume, 0),
    min: parts.length
      ? ([0, 1, 2].map((i) => Math.min(...parts.map((p) => p.min[i]!))) as Vec3)
      : [-1, -1, -1],
    max: parts.length
      ? ([0, 1, 2].map((i) => Math.max(...parts.map((p) => p.max[i]!))) as Vec3)
      : [1, 1, 1],
  };
  const entry = {
    project,
    result: canReuse ? freezeResourceBuilding(result, reused) : freezeBuilding(result),
  };
  buildings.set(input, entry);
  recentBuilding = new WeakRef(entry);
  return entry.result;
}
